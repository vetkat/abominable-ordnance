# Cryo ward Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use
> superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task-by-task. Steps
> use checkbox (`- [ ]`) syntax for tracking.

**Goal:** With Space Age, a cryo ward made on Aquilo from lithium plates
and cold fluoroketone slows the spoiling of a lifecycle item 100 times.
The base ward stays as it is in both modes.

**Architecture:** The stasis builder, `core/prototypes/stasis.lua`,
takes a list of ward kinds. A ward kind is a plain data table: names,
factor, tint, the ward recipe and the research. The builder makes each
kind's ward, warded items, recipes and research, and the `ao-stasis`
subgroup once.

A pure function, `Stasis.kinds`, picks the kinds from facts that
`data.lua` gathers. `data.lua` stays a thin shell. There is no runtime
code.

**Tech Stack:** Factorio 2.1.24 Lua, busted for the core, factorio-test
for the in-game tests, `make check` for all gates.

**Spec:** `docs/superpowers/specs/2026-11-15-cryo-ward-design.md`
(approved by EGM. The maintainer accepted every default in "Calls for
the maintainer" on 2026-11-15).

In every step, `BUSTED` stands for
`lua5.2 /usr/lib/luarocks/rocks-5.2/busted/2.3.0-1/bin/busted --lua=lua5.2`.

## Global Constraints

- The target is Factorio 2.1, with and without Space Age. The cryo ward
  exists only with Space Age.
- Every number lives in `core/balance.lua`. The cryo numbers go under
  `Balance.cryo`.
- A core module never names `game`, `storage`, `script`, `defines`,
  `prototypes`, `rendering`, `helpers`, `remote`, `commands`, `rcon`,
  `settings`, `mods` or `data`. `spec/architecture_spec.lua` enforces
  this, also for table keys.
- No file in `core/`, `shell/` or `control.lua` uses `math.random`.
- Every new item, recipe and technology gets a name in
  `locale/en/locale.cfg`. `spec/locale_spec.lua` enforces this.
- A change to mod code gets a line under Unreleased in `CHANGELOG.md`.
- Commits: `make format`, `git add <files>` (never `-A`), `make stamp`,
  `git commit`, each as its own command. The PR body has the
  `## Review angles` section with all six angles, and the PR uses
  `gh pr merge --auto --squash`.
- Look up each Factorio API name in `.factorio-api/factorio/library/`.
  Pass that path to the search: a search from the repository root skips
  the folder.
- Never stage the `.factorio-api` symlink.
- Cryo factor 100.
- Cryo ward recipe: 2 lithium plates and 10 cold fluoroketone give 1
  cryo ward and 5 hot fluoroketone in 4 s. The category is
  `cryogenics`. One surface condition on `pressure` has `min` and `max`
  both 300.
- The cryo ward recipe sets `auto_recycle = false`,
  `allow_productivity = false` and `allow_quality = false`.
- Every cryo pack and unpack recipe sets `allow_quality = false`,
  `allow_productivity = false`, `maximum_productivity = 0`,
  `allow_decomposition = false`, `allow_as_intermediate = false` and
  `auto_recycle = false`, uses the category `crafting`, takes 1 s and
  sits in the `ao-stasis` subgroup. Unpack sets `main_product`. No
  result sets `always_fresh` or `reset_freshness_on_craft`.
- In-game tests destroy what they create before they assert, give every
  poll an upper bound, and use positions on the row `y = 4000`, from
  `x = 4000` on. No other test file uses that row. The Aquilo test uses

local Helpers = require("__abominable-ordnance__/tests/ingame/helpers")
local Names = require("core.names")

local ROCKET = "ao-biter-larva-explosive"
local WARDED = "ao-warded-biter-larva-explosive"
local CRYO_WARD = "probe-cryo-ward"
local CRYO_WARDED = "probe-cryo-warded-rocket"

local surface = function()
  return game.surfaces[1]
end

local function set_progress(stack, p)
  local total = prototypes.item[stack.name].get_spoil_ticks(stack.quality)
  stack.spoil_tick = game.tick + math.ceil((1 - p) * total)
end

local function near(a, b)
  return math.abs(a - b) <= 0.02
end

local function assembler(position, recipe, inputs)
  Helpers.arena(surface(), position)
  Helpers.research(true, Names.STASIS_TECH)
  local machine = Helpers.powered_machine(surface(), position, "assembling-machine-3")
  machine.set_recipe(recipe)
  local input = machine.get_inventory(defines.inventory.crafter_input)
  for _, stack in ipairs(inputs) do
    input.insert(stack)
  end
  return machine, input
end

local function output(machine)
  return machine.get_inventory(defines.inventory.crafter_output)
end

-- Copies a stack from one inventory to another and clears the source.
-- Returns the progress before the move, so a failure names the step.
local function move(from, to, name)
  local stack = from.find_item_stack(name)
  assert(stack, "no " .. name .. " to move")
  local p = stack.spoil_percent
  assert(to.insert(stack) == 1, "could not move the " .. name)
  stack.clear()
  local moved = to.find_item_stack(name)
  assert(moved and near(moved.spoil_percent, p), "the move changed the progress of " .. name)
  return p
end

tags("space-age")
test("probe: a cryo pack keeps about 40%", function()
  local machine, input = assembler({ 4000, 4500 }, "probe-cryo-pack", {
    { name = ROCKET, count = 1 },
    { name = CRYO_WARD, count = 1 },
  })
  set_progress(input.find_item_stack(ROCKET), 0.4)
  after_ticks(90, function()
    local stack = output(machine).find_item_stack(CRYO_WARDED)
    local p = stack and stack.spoil_percent
    machine.destroy()
    assert(stack, "no cryo-warded rocket")
    log("probe pack: " .. p)
    assert(near(p, 0.4), "cryo-warded rocket at " .. p)
  end)
end)

tags("space-age")
test("probe: a cryo unpack keeps about 40% and gives the ward back", function()
  local machine, input = assembler({ 4040, 4500 }, "probe-cryo-unpack", {
    { name = CRYO_WARDED, count = 1 },
  })
  set_progress(input.find_item_stack(CRYO_WARDED), 0.4)
  after_ticks(90, function()
    local out = output(machine)
    local rocket = out.find_item_stack(ROCKET)
    local wards = out.get_item_count(CRYO_WARD)
    local p = rocket and rocket.spoil_percent
    machine.destroy()
    assert(rocket, "no rocket")
    log("probe unpack: " .. p .. ", wards " .. wards)
    assert(near(p, 0.4), "rocket at " .. tostring(p))
    assert(wards == 1, wards .. " wards")
  end)
end)

tags("space-age")
test("probe: the hand-off keeps about 40%", function()
  local a, a_in = assembler({ 4080, 4500 }, "ao-ward-biter-larva-explosive", {
    { name = ROCKET, count = 1 },
    { name = Names.WARD, count = 1 },

  local b, b_in = assembler({ 4120, 4500 }, "ao-unward-biter-larva-explosive", {})

        end

        log("probe hand-off: " .. p)
        assert(near(p, 0.4), "cryo-warded rocket at " .. p)
      end)
    end)
  end)
end)

## Review Focus

- **Another mod removes an Aquilo prototype.** The base ward must still
  load, and the log must say which prototype is missing. Task 4 pins
  `Stasis.kinds` with one missing key, and Task 6 wires the log line.
- **A player on Nauvis with barrels of cold fluoroketone.** The ward
  recipe must not run there. Task 7 pins it with a plant on Nauvis.
- **A cryo-warded item that spoils.** It must hatch like its plain item,
  on the same force. Task 4 pins the copied trigger, and Task 7 spoils a
  cryo-warded rocket in the game.
- **A quality egg.** An uncommon egg needs an uncommon cryo ward, so a
  player must still be able to set a cryo pack recipe to a quality.
  Task 4 pins that no cryo recipe sets `can_set_quality`, which the
  slice 5 probe showed keeps a quality recipe settable with
  `allow_quality = false`.
- **The spoil time of a cryo-warded egg.** 100 times 30 minutes is
  10 800 000 ticks, which must fit a `uint32`. Task 4 pins that every
  cryo-warded `spoil_ticks` is below 2^32.

## Decisions the plan makes

The spec leaves these open, or its default does not work as written.
Each one is a small change that the maintainer can reverse.

1. **Cryo tint.** The spec's default, `{ r = 0.6, g = 0.85, b = 1 }`,
   is within 0.05 of the base ward tint `{ r = 0.55, g = 0.8, b = 1 }`,
   so the two icons would look the same. The plan uses a deeper, colder
   blue, `{ r = 0.3, g = 0.55, b = 1, a = 1 }`, in `Stasis.CRYO_TINT`.
   One line changes it back.
2. **Two PRs, not one.** The spec says one PR. The plan splits the
   builder refactor (no game change) from the cryo ward, so each PR
   stays small and the refactor gets its own review.
3. **Kind name functions take the kind first.** `Stasis.warded_name`,
   `Stasis.pack_recipe` and `Stasis.unpack_recipe` become
   `(kind, name)`. The two existing callers change with them.
4. **The facts for `Stasis.kinds`.** `facts.has` is a set of
   `"type/name"` keys, the same shape as `Stubs.by_name` uses. The cryo
   kind lists what it needs in `Stasis.CRYO.needs`, so `data.lua` does
   not repeat the names.
5. **The research name lives on the kind.** `kind.research.name` names
   the technology. For both kinds it equals the ward item name, as the
   spec says.
6. **The hand-off test uses an assembling machine 3.** A larva rocket at
   40% gains about 0.3% per second while plain. Faster crafts keep the
   plain time under 3 s, so the result stays within the spec's 2
   points.
7. **Heat on Aquilo.** The spec places a heating tower next to the
   plant. The plan puts two heat pipes between the tower and the plant,
   because the tower's heat connections sit one tile out on each side.
   The test waits until `plant.frozen` is false, with an upper bound.
8. **Word list.** `Aquilo`, `cryo` and `fluoroketone` move from the
   docs-only entries in `.ste-allow` into `cspell.json`, because the
   code, the locale and the tests now use them.
9. **In-game tests for Space Age only.** Every cryo test carries
   `tags("space-age")`. One base-only test pins that the cryo ward does
   not exist without Space Age.

## Ledger rulings made while building

- **The Nauvis check pins data, not a craft.** The engine checks a
  recipe's surface conditions when a player picks the recipe, not when a
  script sets it. A cryogenic plant on Nauvis with `set_recipe` still
  made a cryo ward in the game (1 craft in 300 ticks, status
  `item_ingredient_shortage` after). So Task 7 pins what the picker
  reads: the recipe has one condition on `pressure` that Aquilo meets
  and Nauvis does not. The Review Focus item "the ward recipe must not
  run there" holds for a player, not for a script. Cost if wrong: a
  script or a blueprint can run the recipe on another planet.
- **A control test for the heat.** "The unheated plant freezes" proves
  that the heat test waits for something. The plan's layout (two heat
  pipes between tower and plant, tower 6 tiles south) worked at the
  first run: the plant unfroze and made the ward. No layout change.
- **The spelling words moved in Task 4, not Task 5.** The code of Task 4
  uses `cryo`, and the commit hook runs the text gate. The three
  `cspell.json` words moved with that commit. The `.ste-allow` lines
  went in Task 5 as planned.
- **Review round 1: more needs, and the pressure from the planet.**
  `Stasis.CRYO.needs` now also lists `item/space-science-pack`,
  `item/cryogenic-science-pack` (they are type `item` in 2.1, not
  `tool`), `assembling-machine/cryogenic-plant` and `planet/aquilo`.
  `data.lua` reads Aquilo's pressure from the planet and passes it as
  `facts.pressure`. `Stasis.cryo_kind(pressure)` builds the kind, and
  `Balance.cryo.pressure` (300) is the fallback. This replaces
  `Balance.cryo.surface_conditions` in Tasks 3 and 4.

  The heat test now also asserts that the heated plant is still
  unfrozen 600 ticks after the craft starts, the window in which the
  unheated control freezes. `military-science-pack` stays out of the
  needs: the base ward uses it too and asserts it nowhere.
- **The kind comment moved to the base kind.** The base kind's `needs`
  comment said the picker "will" drop kinds. It now says that
  `Stasis.kinds` does.

-- Builds the stasis wards as plain tables. data.lua passes the result to
-- data:extend. This module reads no game globals: only its parameters
-- and other core modules.
--
-- A ward kind is the plain data that tells one ward from another: its
-- names, its factor, its tint, its recipe and its research. The builder
-- reads a kind and never changes it.
local Balance = require("core.balance")
local Common = require("core.prototypes.common")
local Names = require("core.names")
local Payloads = require("core.payloads")
local Species = require("core.species")

local Stasis = {}

Stasis.SUBGROUP = "ao-stasis"
Stasis.WARD_ICON = "__base__/graphics/icons/plastic-bar.png"
Stasis.WARD_TINT = { r = 0.55, g = 0.8, b = 1, a = 1 }

local PACKS = {
  "automation-science-pack",
  "logistic-science-pack",
  "military-science-pack",
  "chemical-science-pack",
}

local function item(name, amount)
  return { type = "item", name = name, amount = amount }
end

-- The base ward of slice 5, in both modes.
local function base_kind()
  local w = Balance.stasis.ward
  return {
    name = Names.WARD,
    factor = Balance.stasis.factor,
    pack_time = Balance.stasis.pack_time,
    warded = Names.WARDED,
    pack = Names.PACK,
    unpack = Names.UNPACK,
    tint = Stasis.WARD_TINT,
    stack = w.stack,
    recipe = {
      categories = { "crafting" },
      time = w.time,
      ingredients = {
        item("plastic-bar", w.plastic),
        item("steel-plate", w.steel),
        item("electronic-circuit", w.circuit),
      },
      results = { item(Names.WARD, 1) },
    },
    research = {
      name = Names.STASIS_TECH,
      prerequisites = { Names.ORDNANCE_TECH },
      count = Balance.stasis.research.count,
      time = Balance.stasis.research.time,
      packs = PACKS,
    },
    -- The prototypes of other mods that the kind needs. The base ward
    -- needs none here: data.lua asserts its vanilla ingredients.
    needs = {},
  }
end

Stasis.BASE = base_kind()

-- The 10 lifecycle items of version 1: the eggs, the larvae and the
-- larva rockets. A version 2 item joins only by a change here.
function Stasis.lifecycle_names()
  local names = {}
  for _, species in ipairs(Species.ALL) do
    names[#names + 1] = species.egg
    names[#names + 1] = species.larva
  end
  for _, species in ipairs(Species.ALL) do
    for _, payload in ipairs(Payloads.ALL) do
      names[#names + 1] = Payloads.loaded_name(species, payload)
    end
  end
  return names
end

local function suffix(name)
  return (name:gsub("^ao%-", ""))
end

function Stasis.warded_name(kind, name)
  return kind.warded .. suffix(name)
end

function Stasis.pack_recipe(kind, name)
  return kind.pack .. suffix(name)
end

function Stasis.unpack_recipe(kind, name)
  return kind.unpack .. suffix(name)
end

local function ward_icons(kind)
  return Common.icon(Stasis.WARD_ICON, kind.tint)
end

-- A copy of an item's icons with the kind's ward icon in the corner.
local function with_ward(icons, kind)
  local out = Common.copy(icons)
  out[#out + 1] = {
    icon = Stasis.WARD_ICON,
    icon_size = 64,
    scale = 0.25,
    shift = { 8, 8 },
    tint = kind.tint,
  }
  return out
end

-- Pack and unpack give no free quality, no free items and no crafting
-- loops, and the recycler must not undo them.
local function locked(r)
  r.allow_quality = false
  r.allow_productivity = false
  r.maximum_productivity = 0
  r.allow_decomposition = false
  r.allow_as_intermediate = false
  r.subgroup = Stasis.SUBGROUP
  return Common.no_recycling(r)
end

-- Returns a copy of a prototype list in which every item that spoils has
-- an explicit weight. The weight is the rocket lift weight over the stack
-- size, which is the engine's default. Without it, Space Age derives a
-- weight from the item's recipe. A warded item has another recipe, so it
-- would weigh something else. A weight that a prototype sets stays. The
-- input list and its prototypes stay as they are.
function Stasis.weighted(list, lift_weight)
  local out = {}
  for i, p in ipairs(list) do
    if p.spoil_ticks and not p.weight then
      p = Common.copy(p)
      p.weight = lift_weight / p.stack_size
    end
    out[i] = p
  end
  return out
end

-- A warded item is a new plain item. It never copies the source, so a
-- larva rocket's ammo fields stay out.
local function warded(source, kind)
  local p = Common.item(
    Stasis.warded_name(kind, source.name),
    with_ward(source.icons, kind),
    Stasis.SUBGROUP,
    source.stack_size
  )
  p.spoil_ticks = source.spoil_ticks * kind.factor
  p.spoil_to_trigger_result = Common.copy(source.spoil_to_trigger_result)
  p.weight = source.weight
  p.auto_recycle = false
  return p
end

-- Pack and unpack sort by their warded item's order, pack first.
local function pack(source, ward, kind)
  local r = locked(
    Common.recipe(
      Stasis.pack_recipe(kind, source.name),
      { "crafting" },
      kind.pack_time,
      { item(source.name, 1), item(kind.name, 1) },
      { item(ward.name, 1) }
    )
  )
  r.order = ward.order .. "-a"
  return r
end

local function unpack(source, ward, kind)
  local r = locked(
    Common.recipe(
      Stasis.unpack_recipe(kind, source.name),
      { "crafting" },
      kind.pack_time,
      { item(ward.name, 1) },
      { item(source.name, 1), item(kind.name, 1) }
    )
  )
  r.order = ward.order .. "-b"
  r.main_product = source.name
  r.icons = with_ward(source.icons, kind)
  return r
end

-- The subgroup for every stasis prototype, of every kind.
local function subgroup()
  return {
    type = "item-subgroup",
    name = Stasis.SUBGROUP,
    group = "intermediate-products",
    order = "z[ao]-stasis",
  }
end

-- A kind's ward item and the recipe that makes it.
local function ward_kit(kind)
  local k = kind.recipe
  local recipe = Common.recipe(
    kind.name,
    Common.copy(k.categories),
    k.time,
    Common.copy(k.ingredients),
    Common.copy(k.results)
  )
  recipe.subgroup = Stasis.SUBGROUP
  local ward_item = Common.item(kind.name, ward_icons(kind), Stasis.SUBGROUP, kind.stack)
  recipe.order = ward_item.order
  return ward_item, recipe
end

-- One kind's ward, its warded items, their recipes and its research.
-- by_name indexes the lifecycle items. It raises an error for a
-- lifecycle item that by_name lacks, because the pack and unpack
-- recipes need it.
local function kind_prototypes(kind, by_name)
  local ward_item, ward_recipe = ward_kit(kind)
  local out = { ward_item, ward_recipe }
  local unlocks = { ward_recipe.name }
  for _, name in ipairs(Stasis.lifecycle_names()) do
    local source = by_name[name]
    if not source then
      error("stasis: the sources lack the lifecycle item " .. name)
    end
    local w = warded(source, kind)
    local p, u = pack(source, w, kind), unpack(source, w, kind)
    Common.append(out, { w, p, u })
    Common.append(unlocks, { p.name, u.name })
  end
  local research = kind.research
  out[#out + 1] = Common.technology({
    name = research.name,
    icons = ward_icons(kind),
    prerequisites = Common.copy(research.prerequisites),
    count = research.count,
    time = research.time,
    packs = Common.copy(research.packs),
    unlocks = unlocks,
  })
  return out
end

-- Returns the stasis prototypes of every kind in kinds, with the
-- subgroup once. sources is a list of prototypes that holds the
-- lifecycle items, as Lifecycle.prototypes and Ordnance.prototypes
-- return them.
function Stasis.prototypes(sources, kinds)
  local by_name = {}
  for _, p in ipairs(sources) do
    if p.spoil_ticks then
      by_name[p.name] = p
    end
  end
  local out = { subgroup() }
  for _, kind in ipairs(kinds) do
    Common.append(out, kind_prototypes(kind, by_name))
  end
  return out
end

return Stasis

  for tx = x - 12, x + 12 do
    for ty = y - 12, y + 12 do

    end

-- Calls fn once the plant is no longer frozen, within tries seconds.
local function once_warm(plant, tries, fn)
  if not plant.frozen then
    return fn()
  end
  assert(tries > 0, "the cryogenic plant stayed frozen")
  after_ticks(60, function()
    once_warm(plant, tries - 1, fn)
  end)
end
