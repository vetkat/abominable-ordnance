# Slice 5 (stasis ward) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use
> superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task-by-task. Steps
> use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Each lifecycle item gets a warded version that spoils 20 times
more slowly. Plain recipes pack and unpack it, and the Stasis ward
research unlocks them.

**Architecture:** One pure builder, `core/prototypes/stasis.lua`, takes
the prototypes that `Lifecycle.prototypes` and `Ordnance.prototypes`
already build. It picks the 10 version 1 lifecycle items from an explicit
list, and returns the ward, the warded items, the recipes, a subgroup and
the research. `data.lua` stays a thin shell. There is no runtime code.

**Tech Stack:** Factorio 2.1.24 Lua, busted for the core, factorio-test
for the in-game tests, `make check` for all gates.

**Spec:** `docs/superpowers/specs/2026-11-15-stasis-design.md` (three EGM
rounds, round 3 clean. The maintainer allowed the overnight build once
EGM agrees).

In every step, `BUSTED` stands for
`lua5.2 /usr/lib/luarocks/rocks-5.2/busted/2.3.0-1/bin/busted --lua=lua5.2`.

## Global Constraints

- The target is Factorio 2.1, with and without Space Age.
- Every number lives in `core/balance.lua`.
- A core module never names `game`, `storage`, `script`, `defines`,
  `prototypes`, `rendering`, `helpers`, `remote`, `commands`, `rcon`,
  `settings`, `mods` or `data`. `spec/architecture_spec.lua` enforces
  this, also for table keys. Write a key with a global's name in quotes
  or pick another name.
- No file in `core/`, `shell/` or `control.lua` uses `math.random`.
- Every new item, recipe and technology gets a name in
  `locale/en/locale.cfg`. `spec/locale_spec.lua` enforces this.
- A change to mod code gets a line under Unreleased in `CHANGELOG.md`.
- Commits: `make format`, `git add`, `make stamp`, `git commit`, each as
  its own command. The PR body has the `## Review angles` section, and
  the PR uses `gh pr merge --auto --squash`.
- Look up each Factorio API name in `.factorio-api/factorio/library/`.
- Warded spoil time is 20 times the item's own time.
- Pack and unpack set `allow_quality = false`,
  `allow_productivity = false`, `maximum_productivity = 0`,
  `allow_decomposition = false`, `allow_as_intermediate = false` and
  `auto_recycle = false`. No result sets `always_fresh` or
  `reset_freshness_on_craft`.

## Review Focus

- **An item at a quality other than normal.** An uncommon egg needs an
  uncommon ward, and unpacking keeps both uncommon. Task 4 pins it.
- **A warded item that spoils in a machine's input.** It hatches next to
  the machine, like a plain item. Task 4 pins the hatch path through a
  chest, the same trigger.
- **Another mod removes plastic, steel or the circuit.** The data stage
  stops with a clear message. Task 3 asserts each ingredient.
- **A lifecycle item missing from the sources.** The builder skips it and
  makes no recipe that names a missing item. Task 2 pins it.
- **Hand crafting.** A player packs by hand, and the result keeps the
  spoil progress. The probe covers it, and Task 4 covers the assembler.

```lua
local Stasis = require("core.prototypes.stasis")
local Stubs = require("spec.support.stubs")

local function sources()
  local list = Lifecycle.prototypes(Stubs.lifecycle_opts(false))
  for _, p in ipairs((Ordnance.prototypes(Stubs.ordnance_opts()))) do
    list[#list + 1] = p
  end
```

Create `core/prototypes/stasis.lua`:

```lua
-- Builds the stasis ward of slice 5 as plain tables. data.lua passes the
-- result to data:extend. This module reads no game globals: only its
-- parameters and other core modules.
local Balance = require("core.balance")
local Common = require("core.prototypes.common")
local Names = require("core.names")
local Payloads = require("core.payloads")
local Species = require("core.species")

local Stasis = {}

Stasis.SUBGROUP = "ao-stasis"
Stasis.WARD_ICON = "__base__/graphics/icons/plastic-bar.png"
Stasis.WARD_TINT = { r = 0.55, g = 0.8, b = 1, a = 1 }

local PACKS = {
  "automation-science-pack",
  "logistic-science-pack",
  "military-science-pack",
  "chemical-science-pack",
}

-- The 10 lifecycle items of version 1: the eggs, the larvae and the
-- larva rockets. A version 2 item joins only by a change here.
function Stasis.lifecycle_names()
  local names = {}
  for _, species in ipairs(Species.ALL) do
    names[#names + 1] = species.egg
    names[#names + 1] = species.larva
  end
  for _, species in ipairs(Species.ALL) do
    for _, payload in ipairs(Payloads.ALL) do
      names[#names + 1] = Payloads.loaded_name(species, payload)
    end
  end
  return names
end

local function suffix(name)
  return (name:gsub("^ao%-", ""))
end

function Stasis.warded_name(name)
  return Names.WARDED .. suffix(name)
end

function Stasis.pack_recipe(name)
  return Names.PACK .. suffix(name)
end

function Stasis.unpack_recipe(name)
  return Names.UNPACK .. suffix(name)
end

-- A copy of an item's icons with the ward icon in the corner.
local function with_ward(icons)
  local out = Common.copy(icons)
  out[#out + 1] = {
    icon = Stasis.WARD_ICON,
    icon_size = 64,
    scale = 0.25,
    shift = { 8, 8 },
    tint = Stasis.WARD_TINT,
  }
  return out
end

-- Pack and unpack give no free quality, no free items and no crafting
-- loops, and the recycler must not undo them.
local function locked(r)
  r.allow_quality = false
  r.allow_productivity = false
  r.maximum_productivity = 0
  r.allow_decomposition = false
  r.allow_as_intermediate = false
  r.subgroup = Stasis.SUBGROUP
  return Common.no_recycling(r)
end

-- A warded item is a new plain item. It never copies the source, so a
-- larva rocket's ammo fields stay out.
local function warded(source)
  local p = Common.item(
    Stasis.warded_name(source.name),
    with_ward(source.icons),
    Stasis.SUBGROUP,
    source.stack_size
  )
  p.spoil_ticks = source.spoil_ticks * Balance.stasis.factor
  p.spoil_to_trigger_result = Common.copy(source.spoil_to_trigger_result)
  p.auto_recycle = false
  return p
end

local function item(name, amount)
  return { type = "item", name = name, amount = amount }
end

local function pack(source, ward)
  return locked(
    Common.recipe(
      Stasis.pack_recipe(source.name),
      { "crafting" },
      Balance.stasis.pack_time,
      { item(source.name, 1), item(Names.WARD, 1) },
      { item(ward.name, 1) }
    )
  )
end

local function unpack(source, ward)
  local r = locked(
    Common.recipe(
      Stasis.unpack_recipe(source.name),
      { "crafting" },
      Balance.stasis.pack_time,
      { item(ward.name, 1) },
      { item(source.name, 1), item(Names.WARD, 1) }
    )
  )
  r.main_product = source.name
  r.icons = with_ward(source.icons)
  return r
end

-- The ward itself, its recipe and the subgroup for every stasis
-- prototype.
local function ward_kit()
  local w = Balance.stasis.ward
  local icons = Common.icon(Stasis.WARD_ICON, Stasis.WARD_TINT)
  local subgroup = {
    type = "item-subgroup",
    name = Stasis.SUBGROUP,
    group = "intermediate-products",
    order = "z[ao]-stasis",
  }
  local recipe = Common.recipe(Names.WARD, { "crafting" }, w.time, {
    item("plastic-bar", w.plastic),
    item("steel-plate", w.steel),
    item("electronic-circuit", w.circuit),
  }, { item(Names.WARD, 1) })
  recipe.subgroup = Stasis.SUBGROUP
  return subgroup, Common.item(Names.WARD, icons, Stasis.SUBGROUP, w.stack), recipe
end

-- Returns the stasis prototypes. sources is a list of prototypes that
-- holds the lifecycle items, as Lifecycle.prototypes and
-- Ordnance.prototypes return them. A lifecycle item that sources lack
-- gets no warded version.
function Stasis.prototypes(sources)
  local by_name = {}
  for _, p in ipairs(sources) do
    if p.spoil_ticks then
      by_name[p.name] = p
    end
  end
  local subgroup, ward_item, ward_recipe = ward_kit()
  local out = { subgroup, ward_item, ward_recipe }
  local unlocks = { ward_recipe.name }
  for _, name in ipairs(Stasis.lifecycle_names()) do
    local source = by_name[name]
    if source then
      local w = warded(source)
      local p, u = pack(source, w), unpack(source, w)
      Common.append(out, { w, p, u })
      Common.append(unlocks, { p.name, u.name })
    end
  end
  out[#out + 1] = Common.technology({
    name = Names.STASIS_TECH,
    icons = Common.icon(Stasis.WARD_ICON, Stasis.WARD_TINT),
    prerequisites = { Names.ORDNANCE_TECH },
    count = Balance.stasis.research.count,
    time = Balance.stasis.research.time,
    packs = PACKS,
    unlocks = unlocks,
  })
  return out
end

return Stasis
```

- [ ] **Step 1: Write the tests**

Create `tests/ingame/stasis_test.lua`:

```lua
local Helpers = require("__abominable-ordnance__/tests/ingame/helpers")
local Names = require("core.names")
local Species = require("core.species")

local BITER = Species.ALL[1]
local ROCKET = "ao-biter-larva-explosive"
local WARDED_ROCKET = "ao-warded-biter-larva-explosive"

local surface = function()
  return game.surfaces[1]
end

local function unit_names(species)
  local names = {}
  for _, unit in ipairs(species.units) do
    names[#names + 1] = unit[1]
  end
  return names
end

-- Sets a stack's spoil progress p (0 to 1). spoil_tick is an absolute
-- game tick, so the test sets it right after the insert.
local function set_progress(stack, p)
  local total = prototypes.item[stack.name].get_spoil_ticks(stack.quality)
  stack.spoil_tick = game.tick + math.ceil((1 - p) * total)
end

local function near(a, b)
  return math.abs(a - b) <= 0.02
end

-- An assembler with power that runs one recipe, with its input filled.
local function assembler(position, recipe, inputs)
  Helpers.arena(surface(), position)
  Helpers.research(true, Names.STASIS_TECH)
  local machine = Helpers.powered_machine(surface(), position, "assembling-machine-2")
  machine.set_recipe(recipe)
  local input = machine.get_inventory(defines.inventory.assembling_machine_input)
  for _, stack in ipairs(inputs) do
    input.insert(stack)
  end
  return machine, input
end

local function output(machine)
  return machine.get_inventory(defines.inventory.assembling_machine_output)
end

before_each(function()
  game.forces["enemy"].set_evolution_factor(0, surface())
end)

test("an assembler packs a rocket at 40% into a warded rocket at about 40%", function()
  local machine, input = assembler({ 2000, 500 }, "ao-ward-biter-larva-explosive", {
    { name = ROCKET, count = 1 },
    { name = Names.WARD, count = 1 },
  })
  set_progress(input.find_item_stack(ROCKET), 0.4)
  after_ticks(180, function()
    local stack = output(machine).find_item_stack(WARDED_ROCKET)
    assert(stack, "no warded rocket")
    local p = stack.spoil_percent
    machine.destroy()
    assert(near(p, 0.4), "warded rocket at " .. p)
  end)
end)

test("unpacking a warded rocket at 40% gives a rocket at about 40% and a ward", function()
  local machine, input = assembler({ 2040, 500 }, "ao-unward-biter-larva-explosive", {
    { name = WARDED_ROCKET, count = 1 },
  })
  set_progress(input.find_item_stack(WARDED_ROCKET), 0.4)
  after_ticks(180, function()
    local out = output(machine)
    local rocket = out.find_item_stack(ROCKET)
    local wards = out.get_item_count(Names.WARD)
    local p = rocket and rocket.spoil_percent
    machine.destroy()
    assert(rocket, "no rocket")
    assert(near(p, 0.4), "rocket at " .. tostring(p))
    assert(wards == 1, wards .. " wards")
  end)
end)

test("an uncommon item and an uncommon ward unpack to uncommon ones", function()
  if not prototypes.quality["uncommon"] then
    return
  end
  local machine, input = assembler({ 2080, 500 }, "ao-ward-biter-egg", {
    { name = BITER.egg, count = 1, quality = "uncommon" },
    { name = Names.WARD, count = 1, quality = "uncommon" },
  })
  machine.set_recipe("ao-ward-biter-egg", "uncommon")
  after_ticks(180, function()
    local stack = output(machine).find_item_stack({
      name = "ao-warded-biter-egg",
      quality = "uncommon",
    })
    machine.destroy()
    assert(stack, "no uncommon warded egg")
  end)
end)

test("a warded egg at 99% in a chest is still unspoiled 3 s later", function()
  local position = { 2120, 500 }
  Helpers.arena(surface(), position)
  local chest = surface().create_entity({
    name = "steel-chest",
    position = position,
    force = "player",
  })
  chest.insert({ name = "ao-warded-biter-egg", count = 1 })
  set_progress(chest.get_inventory(defines.inventory.chest)[1], 0.99)
  after_ticks(180, function()
    local count = chest.get_item_count("ao-warded-biter-egg")
    chest.destroy()
    assert(count == 1, "the warded egg spoiled")
  end)
end)

test("a warded rocket that spoils hatches its loaded monster on the tamed force", function()
  local position = { 2160, 500 }
  Helpers.arena(surface(), position)
  Helpers.spoil(Helpers.spill(surface(), position, WARDED_ROCKET, 1))
  after_ticks(2, function()
    local units = Helpers.find_units(surface(), position, { force = Names.TAMED })
    assert(#units == 1, #units .. " tamed units")
    local entry = storage.loaded[units[1].unit_number]
    assert(entry and entry.payloads[1] == "explosive", "no loaded entry")
  end)
end)

test("10 warded biter eggs that spoil hatch one wild biter", function()
  local position = { 2200, 500 }
  Helpers.arena(surface(), position)
  Helpers.spoil(Helpers.spill(surface(), position, "ao-warded-biter-egg", 10))
  after_ticks(2, function()
    local biters = Helpers.count_units(surface(), position, unit_names(BITER))
    assert(biters == 1, biters .. " biters hatched")
  end)
end)
```
